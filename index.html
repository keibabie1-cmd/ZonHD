let RD_TOKEN = '', AD_KEY = ''; // Update fetchConfig to grab all keys async function fetchConfig() { const res = await fetch('/api/config'); const data = await res.json(); API_KEY = data.tmdb_key; RD_TOKEN = data.rd_token; AD_KEY = data.ad_key; fetchContent(); } async function startLuxeStream() { document.getElementById('player-overlay').style.display = 'block'; document.getElementById('ad-shield').style.display = 'flex'; const { type, id, s, e } = currentMediaContext; try { // Step 1: Get the magnet hash from your Render server const hashRes = await fetch(`/get-hash?type=${type}&id=${id}&s=${s}&e=${e}`); const hashData = await hashRes.json(); const hash = hashData.hash; let finalUrl = ''; // Step 2: Try AllDebrid directly from the browser (Your Residential IP) try { const adUpload = await fetch(`https://api.alldebrid.com/v4/magnet/upload?agent=zonhd&apikey=${AD_KEY}&magnets[]=${hash}`); const adUpData = await adUpload.json(); const magId = adUpData.data.magnets[0].id; const adStatus = await fetch(`https://api.alldebrid.com/v4/magnet/status?agent=zonhd&apikey=${AD_KEY}&id=${magId}`); const adStatData = await adStatus.json(); const linkToUnlock = adStatData.data.magnets.links[0].link; const adUnlock = await fetch(`https://api.alldebrid.com/v4/link/unlock?agent=zonhd&apikey=${AD_KEY}&link=${linkToUnlock}`); const adFinal = await adUnlock.json(); finalUrl = adFinal.data.link; } catch (e) { // Step 3: Fallback to Real-Debrid directly from the browser const rdAdd = await fetch(`https://api.real-debrid.com/rest/1.0/torrents/addMagnet`, { method: 'POST', headers: { 'Authorization': `Bearer ${RD_TOKEN}`, 'Content-Type': 'application/x-www-form-urlencoded' }, body: `magnet=magnet:?xt=urn:btih:${hash}` }); const rdAddData = await rdAdd.json(); // Select all files and wait await fetch(`https://api.real-debrid.com/rest/1.0/torrents/selectFiles/${rdAddData.id}`, { method: 'POST', headers: { 'Authorization': `Bearer ${RD_TOKEN}` }, body: `files=all` }); await new Promise(r => setTimeout(r, 3000)); const rdInfo = await fetch(`https://api.real-debrid.com/rest/1.0/torrents/info/${rdAddData.id}`, { headers: { 'Authorization': `Bearer ${RD_TOKEN}` } }); const rdInfoData = await rdInfo.json(); const rdUnrestrict = await fetch(`https://api.real-debrid.com/rest/1.0/unrestrict/link`, { method: 'POST', headers: { 'Authorization': `Bearer ${RD_TOKEN}` }, body: `link=${rdInfoData.links[0]}` }); const rdFinal = await rdUnrestrict.json(); finalUrl = rdFinal.download; } if (finalUrl) { art = new ArtPlayer({ container: '#player-container', url: finalUrl, autoplay: true, fullscreen: true }); art.on('ready', () => document.getElementById('ad-shield').style.display = 'none'); } } catch (err) { document.getElementById('shield-text').innerText = "Streaming Error: Direct Handshake Failed."; } }
